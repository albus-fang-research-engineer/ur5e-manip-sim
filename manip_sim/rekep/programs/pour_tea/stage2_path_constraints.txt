def stage2_path_constraint1(end_effector, keypoints):
    """The robot must still be grasping the teapot (keypoint 0)."""
    return get_grasping_cost_by_keypoint_idx(0)

def stage2_path_constraint2(end_effector, keypoints):
    """The teapot must stay upright to avoid spilling: its up axis, keypoint 2 to keypoint 3, along the z-axis, not against it."""
    up = keypoints[3] - keypoints[2]
    return np.linalg.norm(np.cross(up, np.array([0, 0, 1]))) + max(0.0, -up[2])
